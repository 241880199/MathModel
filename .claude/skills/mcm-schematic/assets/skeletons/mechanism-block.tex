\begin{tikzpicture}[x=1in, y=1in]
  \useasboundingbox (0,0) rectangle (\mcmscfigwidth, 2.45in);

  % —— 液面（先画，储罐只描边 ⇒ 液面透出来）——
  \fill[mcmliquid] (1.90, 1.05) rectangle (3.90, 1.42);
  \draw[mcmdashplain] (1.90, 1.42) -- (3.90, 1.42);
  \node[mcmnote] at (2.90, 1.24) {liquid};
  % 液位示意：一根双向细箭头 + 一个标注（H10 不适用：这是示意轴，不是数据轴）
  \draw[mcmdim] (2.10, 1.05) -- (2.10, 1.42);
  \node[mcmnote, anchor=west] at (2.16, 1.14) {h(t)};

  % —— 储罐本体（只描边的最小块）——
  \node[mcmtank, minimum width=2.00in, minimum height=1.00in] (tank) at (2.90, 1.55) {};

  % —— 流入：细流（Q_in 小）；阀块串在流入管线上 ——
  \node[mcmnode, text width=0.50in, minimum height=0.34in] (valve) at (0.88, 1.70) {Valve};
  \draw[mcmflowthin] (0.12, 1.70) -- (valve.west);
  \draw[mcmflowthin] (valve.east) -- (1.90, 1.70);
  \node[mcmlabel] at (0.50, 1.90) {Q\textsubscript{in}};

  % —— 流出：粗流（Q_out 大）——
  \draw[mcmflow] (3.90, 1.30) -- (5.10, 1.30);
  \node[mcmlabel] at (4.50, 1.50) {Q\textsubscript{out}};

  % —— 量测耦合：储罐 → 控制器（细实线箭头）——
  \draw[mcmarrow] (3.94, 2.05) -- (4.70, 2.05);
  \node[mcmlabel] at (4.32, 2.19) {h(t)};
  \node[mcmnode, text width=1.00in] (ctrl) at (5.40, 2.05) {Controller};

  % —— 控制耦合：控制器 → 阀（虚线 = 控制信号，S2）；绕到全图下方走线，避开储罐 ——
  \draw[mcmdash] (ctrl.south) -- ++(0,-1.60) -| (valve.north);
  \node[mcmlabel] at (3.20, 0.37) {u(t)};
\end{tikzpicture}

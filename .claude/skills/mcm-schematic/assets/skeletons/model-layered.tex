\begin{tikzpicture}[x=1in, y=1in]
  \useasboundingbox (0,0) rectangle (\mcmscfigwidth, 2.00in);

  % 三层的分组框（x 从 0.06 到 6.35；每层高 0.44in）

  \draw[mcmgroup] (0.06, 0.75) rectangle (6.35, 1.19);

  \node[mcmnote, anchor=west] at (0.20, 1.72) {Input layer};
  \node[mcmnote, anchor=west] at (0.20, 0.97) {Hidden layer};
  \node[mcmnote, anchor=west] at (0.20, 0.22) {Output layer};

  % 每层三个单元，**列对齐**：x = 2.30 / 3.90 / 5.50

  \node[mcmnode, text width=1.10in] (a1) at (2.30, 1.72) {Feature 1};
  \node[mcmnode, text width=1.10in] (a2) at (3.90, 1.72) {Feature 2};
  \node[mcmnode, text width=1.10in] (a3) at (5.50, 1.72) {Feature 3};

  \node[mcmnode, text width=1.10in] (b1) at (2.30, 0.97) {Unit 1};
  \node[mcmaccent, text width=1.10in] (b2) at (3.90, 0.97) {Unit 2};
  \node[mcmnode, text width=1.10in] (b3) at (5.50, 0.97) {Unit 3};

  \node[mcmnode, text width=1.10in] (c1) at (2.30, 0.22) {Score};
  \node[mcmnode, text width=1.10in] (c2) at (3.90, 0.22) {Risk};
  \node[mcmnode, text width=1.10in] (c3) at (5.50, 0.22) {Decision};

  % 跨层连线：**按列捆扎**（本骨架只画一次每列的连线；原语兜底时自行决定要不要全连接）
  \mcmedge{a1}{b1}  \mcmedge{a2}{b2}  \mcmedge{a3}{b3}
  \mcmedge{b1}{c1}  \mcmedge{b2}{c2}  \mcmedge{b3}{c3}
\end{tikzpicture}
\end{document}

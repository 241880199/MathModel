\documentclass[12pt]{article}
\input{../schematic-style.tex}
\usepackage[paperwidth=\mcmscfigwidth,paperheight=2.30in,margin=0pt]{geometry}
\pagestyle{empty}
\setlength{\parindent}{0pt}

\begin{document}
\begin{tikzpicture}[x=1in, y=1in]
  \useasboundingbox (0,0) rectangle (\mcmscfigwidth, 2.30in);

  \node[mcmterm, text width=0.88in]              (nIn)  at (0.95, 1.20) {Inflow};
  \node[mcmnode, text width=1.00in, minimum height=0.52in] (nS) at (3.20, 1.20) {Stock};
  \node[mcmterm, text width=0.88in]              (nOut) at (5.45, 1.20) {Outflow};

  % 主流量：实线 = 主流程（S2）；极性徽标标在边的上方
  \mcmedge{nIn}{nS}
  \mcmedge{nS}{nOut}
  \node[mcmlabel] at ($(nIn)!0.5!(nS) + (0,0.11)$) {+};
  \node[mcmlabel] at ($(nS)!0.5!(nOut) + (0,0.11)$) {+};

  % 正反馈环（上）：强调色 + 实线；出点 = nOut.north，入点 = nIn.north
  \draw[mcmarrowa] (nOut.north) -- ++(0,0.55) -| (nIn.north);
  \node[mcmnoteaccent] at (3.20, 2.05) {reinforcing loop \textbf{(+)}};

  % 负反馈环（下）：虚线 = 回环 / 反馈（S2）；出点 = nIn.south，入点 = nOut.south
  \draw[mcmdash] (nIn.south) -- ++(0,-0.55) -| (nOut.south);
  \node[mcmnote] at (3.20, 0.34) {balancing loop \textbf{(-)}};
\end{tikzpicture}
\end{document}
